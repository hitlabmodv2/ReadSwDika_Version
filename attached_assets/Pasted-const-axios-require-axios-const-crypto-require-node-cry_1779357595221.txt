  })

}

async function processBatch(data, func) {
  const key = Object.keys(data);
  const res = await Promise.allSettled(key.map(k => func(k, data[k])));
  return Object.fromEntries(res.map((res, i) => [key[i], res.status === 'fulfilled' ?  res.value : null]));
};

async function process(audioBuffer) {
  try {
    const token = await getToken()
    const fileName = 'mymusik'
    
    // get signed url to upload
    const getSigned = await GraphQL({
      type: 'upload',
      vars: {
        input: fileName,
        type: 'FILESYSTEM'
      }
    }, token)
    
    await axios.put(getSigned.uploadFile.signedUrl, audioBuffer, {
      headers: {
        'content-type': 'audio/mpeg',
        'content-length': String(audioBuffer.length)
      }
    })
    
    // prepare task
    const createTaskParams = await GraphQL({
      type: 'createTask',
      vars: {}
    }, token)
    
    const taskId = createTaskParams.createTaskV2
    if (!taskId) throw new Error('Taks Id is not found')
    
    // linked task to file
    await GraphQL({
      type: 'linkTask',
      vars: {
        taskId: taskId,
        file: {
          provider: 'FILESYSTEM',
          input: fileName,
          tempLocation: getSigned.uploadFile.tempLocation,
          name: fileName,
          type: 'AUDIO'
        },
        operations: [{
          name: 'BEATSCHORDS_A',
          params: {}
        }, {
          name: 'SEPARATE_B',
          params: {
            type: 'vocals-drums-bass-other' // locked on this process
          }
        }]
      }
    }, token)
    
    // polling task
    let taskResult = {}
    while (true) {
      const res = await GraphQL({
        type: 'trackQuery',
        vars: { id: taskId }
      }, token)
      
      taskResult = res.track
      if (res.track.operations.every(p => p.status === 'COMPLETED')) break;
      
      await new Promise(resolve => setTimeout(resolve, 2500))
    }
    
    const outs = taskResult.operations.find(p => p.name === 'SEPARATE_B').result
    const beats = taskResult.operations.find(p => p.name === 'BEATSCHORDS_A').result
    const meta = taskResult.operations.find(p => p.name === 'FILEMETADATA_A').result

    const files = await processBatch(outs.files., async (a, b) => {
      return DLService(b, token, true)
    })
    const file2 = await processBatch(beats.files, async (a, b) => {
      if (b.includes('.json')) {
        return DLService(b, token, false)
      } else {
        return DLService(b, token, true)
      }
    })
    
    // waiting to delete user temporary
    await new Promise(resolve => setTimeout(resolve, 2000));
    await GraphQL({
      type: 'deleteUser',
      vars: { reason: 'something-else' }
    }, token)
    
    return {
      files: files,
      beats: {
        ...beats,
        files: file2
      },
      meta: meta,
      summary: taskResult.summary.v1.original
    }
  } catch (error) {
    throw error
  }
}
